enable_language(Swift)
set(nf_swift_arch "${CMAKE_OSX_ARCHITECTURES}")
if(NOT nf_swift_arch)
  set(nf_swift_arch "${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(nf_swift_deployment "${CMAKE_OSX_DEPLOYMENT_TARGET}")
if(NOT nf_swift_deployment)
  set(nf_swift_deployment "15.0")
endif()
add_library(nf_native_worker STATIC KernelProtocol.swift KernelWorker.swift KernelListener.swift)
set_target_properties(nf_native_worker PROPERTIES Swift_LANGUAGE_VERSION 6 Swift_MODULE_NAME NightfallKernelWorker Swift_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
target_include_directories(nf_native_worker PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/modules" "${CMAKE_CURRENT_SOURCE_DIR}/../bridge")
target_compile_options(nf_native_worker PUBLIC -target "${nf_swift_arch}-apple-macosx${nf_swift_deployment}" PRIVATE -strict-concurrency=complete -warnings-as-errors -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
target_link_libraries(nf_native_worker PRIVATE nf_kernel_bridge)
option(NF_BUILD_XPC_BUNDLE "Build the embedded development service and bundle probes" ON)
if(NF_BUILD_XPC_BUNDLE)
  add_executable(nf_kernel_service MACOSX_BUNDLE ServiceMain.swift)
  target_link_libraries(nf_kernel_service PRIVATE nf_native_worker)
  target_compile_options(nf_kernel_service PRIVATE -parse-as-library -swift-version 6 -strict-concurrency=complete -warnings-as-errors -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
  set_target_properties(nf_kernel_service PROPERTIES BUNDLE_EXTENSION xpc MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${NF_OCCT_INSTALL_DIR}/lib")
endif()
if(NF_BUILD_TESTS)
  target_compile_options(nf_native_worker PRIVATE -enable-testing)
  add_executable(nf_native_worker_tests ../tests/worker_tests.swift)
  target_compile_options(nf_native_worker_tests PRIVATE -parse-as-library -swift-version 6 -strict-concurrency=complete -warnings-as-errors -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
  target_link_libraries(nf_native_worker_tests PRIVATE nf_native_worker)
  set_target_properties(nf_native_worker_tests PROPERTIES BUILD_RPATH "${NF_OCCT_INSTALL_DIR}/lib")
  foreach(case limits admission lru cache residency factory watchdog factory-deadline races references xpc termination)
    add_test(NAME native.${case} COMMAND nf_native_worker_tests ${case} "${PROJECT_SOURCE_DIR}/examples/parts")
    set_tests_properties(native.${case} PROPERTIES TIMEOUT 60)
  endforeach()
  if(NF_BUILD_XPC_BUNDLE)
    add_subdirectory("${PROJECT_SOURCE_DIR}/app/tests" "${PROJECT_BINARY_DIR}/app/tests")
  endif()
endif()
if(NF_BUILD_XPC_BUNDLE)
  add_subdirectory("${PROJECT_SOURCE_DIR}/app" "${PROJECT_BINARY_DIR}/app")
endif()
