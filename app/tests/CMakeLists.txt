find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(protocol "${PROJECT_SOURCE_DIR}/services/kernel/swift/KernelProtocol.swift")
set(service_plist "${PROJECT_SOURCE_DIR}/services/kernel/swift/Info.plist.in")
add_executable(nf_fault_service MACOSX_BUNDLE FaultService.swift FaultProtocol.swift "${protocol}")
target_link_libraries(nf_fault_service PRIVATE nf_native_worker)
target_compile_options(nf_fault_service PRIVATE -parse-as-library -swift-version 6 -strict-concurrency=complete -warnings-as-errors -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
set_target_properties(nf_fault_service PROPERTIES BUNDLE_EXTENSION xpc MACOSX_BUNDLE_INFO_PLIST "${service_plist}" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${NF_OCCT_INSTALL_DIR}/lib")
foreach(kind probe fault_probe)
  add_executable(nf_xpc_${kind} MACOSX_BUNDLE XPCProbe.swift GeometryFixture.swift FaultProtocol.swift ClientIntegration.swift RendererTests.swift PreviewTests.swift HistoryTests.swift DraftTests.swift SheetTests.swift ../PartProgram.swift ../DocumentSheet.swift ../PreviewDocument.swift ../KernelClient.swift ../DisplayMesh.swift ../MetalRenderer.swift ../OrbitView.swift "${protocol}")
  target_compile_options(nf_xpc_${kind} PRIVATE -parse-as-library -swift-version 6 -strict-concurrency=complete -warnings-as-errors -target "${nf_swift_arch}-apple-macosx${nf_swift_deployment}" -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
  set_target_properties(nf_xpc_${kind} PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "dev.nightfall.test.${kind}" MACOSX_BUNDLE_BUNDLE_VERSION 1 MACOSX_BUNDLE_SHORT_VERSION_STRING "0.0.1")
  if(kind STREQUAL "probe")
    set(service nf_kernel_service)
  else()
    set(service nf_fault_service)
  endif()
  add_dependencies(nf_xpc_${kind} ${service})
  add_custom_command(TARGET nf_xpc_${kind} POST_BUILD
    COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/embed_service.py" "$<TARGET_BUNDLE_DIR:nf_xpc_${kind}>" "$<TARGET_BUNDLE_DIR:${service}>" "${NF_OCCT_INSTALL_DIR}"
    VERBATIM)
  set_property(TARGET nf_xpc_${kind} APPEND PROPERTY LINK_DEPENDS "${PROJECT_SOURCE_DIR}/tools/embed_service.py" "$<TARGET_FILE:${service}>")
endforeach()
foreach(case smoke references restart crash hang client)
  if(case MATCHES "^(crash|hang)$")
    set(host nf_xpc_fault_probe)
  else()
    set(host nf_xpc_probe)
  endif()
  add_test(NAME bundle.${case} COMMAND "$<TARGET_FILE:${host}>" ${case} "${PROJECT_SOURCE_DIR}/examples/parts")
  set_tests_properties(bundle.${case} PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
endforeach()
add_test(NAME bundle.relocation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bundle_tests.py" "$<TARGET_BUNDLE_DIR:nf_xpc_probe>" "$<TARGET_BUNDLE_DIR:nf_xpc_fault_probe>" "${PROJECT_SOURCE_DIR}/examples/parts" "${NF_OCCT_INSTALL_DIR}")
set_tests_properties(bundle.relocation PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
add_test(NAME bundle.render COMMAND "$<TARGET_FILE:nf_xpc_probe>" render "${PROJECT_SOURCE_DIR}/examples/parts" "${PROJECT_BINARY_DIR}/renders")
set_tests_properties(bundle.render PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
add_test(NAME bundle.preview COMMAND "$<TARGET_FILE:nf_xpc_probe>" preview "${PROJECT_SOURCE_DIR}/examples/parts")
set_tests_properties(bundle.preview PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
add_test(NAME bundle.history COMMAND "$<TARGET_FILE:nf_xpc_probe>" history "${PROJECT_SOURCE_DIR}/examples/parts")
set_tests_properties(bundle.history PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
add_test(NAME bundle.drafts COMMAND "$<TARGET_FILE:nf_xpc_probe>" drafts "${PROJECT_SOURCE_DIR}/examples/parts")
set_tests_properties(bundle.drafts PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
add_test(NAME bundle.sheets COMMAND "$<TARGET_FILE:nf_xpc_probe>" sheets)
set_tests_properties(bundle.sheets PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
foreach(suite client client-tsan)
  add_executable(nf_${suite}_tests ClientTests.swift GeometryFixture.swift ../KernelClient.swift "${protocol}")
  target_compile_options(nf_${suite}_tests PRIVATE -parse-as-library -swift-version 6 -strict-concurrency=complete -warnings-as-errors -target "${nf_swift_arch}-apple-macosx${nf_swift_deployment}" -module-cache-path "${CMAKE_CURRENT_BINARY_DIR}/module-cache")
  if(suite STREQUAL "client-tsan")
    target_compile_options(nf_${suite}_tests PRIVATE -sanitize=thread)
  endif()
  foreach(case admission deadline cancellation recovery recovery-success invalid-recovery validation races)
    add_test(NAME ${suite}.${case} COMMAND nf_${suite}_tests ${case})
    set_tests_properties(${suite}.${case} PROPERTIES TIMEOUT 20)
  endforeach()
endforeach()
